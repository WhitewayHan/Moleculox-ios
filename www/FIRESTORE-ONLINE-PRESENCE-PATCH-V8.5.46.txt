function presenceKeys() {
  return ['uid', 'clientId', 'lastSeen', 'expiresAt', 'platform'];
}
function validPresence(data, presenceId) {
  return signedIn()
    && data.keys().hasOnly(presenceKeys())
    && data.uid == request.auth.uid
    && data.clientId is string
    && data.clientId.matches('^[A-Za-z0-9_-]{8,70}$')
    && presenceId == request.auth.uid + '_' + data.clientId
    && data.lastSeen == request.time
    && data.expiresAt is timestamp
    && data.platform in ['ios', 'android', 'web'];
}

match /onlinePresence/{presenceId} {
  allow get: if signedIn();
  allow list: if signedIn()
    && request.query.limit != null
    && request.query.limit <= 500;
  allow create: if validPresence(request.resource.data, presenceId);
  allow update: if validPresence(request.resource.data, presenceId)
    && resource.data.uid == request.auth.uid
    && resource.data.clientId == request.resource.data.clientId;
  allow delete: if signedIn() && resource.data.uid == request.auth.uid;
}
